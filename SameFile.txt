test('Basic API test to fetch customer information', {tag: '@fetch'}, async({request}) => {
     

     // Action of the test - Sending get request
     // ========================================
     const customerResponse = await request.get('URL', 
                headers: {
                'Authorization': 'Bearer token',
                'Content-type': 'application/json'     
                });

     // Parsing the response
     const parsedCustomerResponse = customerResponse.json();
     console.log(JSON.Stringify(parsedCustomerResponse, null, 2));
     expect(parsedCustomerResponse.customerInfo[0].customerName).toBe('Parthiban');
     expect(parsedCustomerResponse.customerName).toBe('Parthiban');
     

     // verify outcome of action step
     // ===============================
     await expect(customerResponse.status()).toBe(200);
    
});
